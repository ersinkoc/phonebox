cmake_minimum_required(VERSION 3.24)
project(PhoneBoxCamera LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PhoneBox Camera requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
add_compile_options(/W4 /EHsc /permissive- /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
add_library(PhoneBoxCamera SHARED MediaSource.cpp FrameReader.cpp CameraHost.cpp PhoneBoxCamera.def)
target_link_libraries(PhoneBoxCamera PRIVATE mfplat mf mfuuid mfsensorgroup ole32 runtimeobject propsys windowscodecs winhttp advapi32)
add_executable(PhoneBoxCameraSetup Setup.cpp)
target_link_libraries(PhoneBoxCameraSetup PRIVATE mfplat mf mfuuid ole32 shell32 advapi32)
add_executable(PhoneBoxCameraTests Tests.cpp FrameReader.cpp)
target_link_libraries(PhoneBoxCameraTests PRIVATE mfplat mf mfuuid mfreadwrite ole32 runtimeobject windowscodecs winhttp ws2_32)
set_target_properties(PhoneBoxCamera PhoneBoxCameraSetup PhoneBoxCameraTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/dist/bin")
enable_testing()
add_test(NAME NativeCamera COMMAND PhoneBoxCameraTests $<TARGET_FILE:PhoneBoxCamera>)
